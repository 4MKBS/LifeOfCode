How it works CSS 
/* Define the container */
.demo {
  container: demo / inline-size;
  resize: horizontal;
}

/* Respond to container width */
@container demo (min-width: 520px) {
  .cards {
    grid-template-columns: repeat(2, 1fr);
  }
}

@container demo (min-width: 780px) {
  .cards {
    grid-template-columns: repeat(3, 1fr);
  }
  .card {
    flex-direction: column;
  }
}
